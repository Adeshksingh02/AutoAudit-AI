# Microsoft ki official Playwright image (isme sab system files pre-installed hain)
FROM mcr.microsoft.com/playwright/python:v1.42.0-jammy

# Working directory set kar rahe hain
WORKDIR /app

# Requirements copy karke install karna
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Baaki saara code copy karna
COPY . .

# Server start karne ka command
CMD ["python", "main.py"]
